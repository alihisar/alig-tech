alig.tech saglik taramasi: calisan tum sorgular
Her sorgu SELECT coalesce(json_agg(t),'[]') FROM (<sorgu>) t icinde,
default_transaction_read_only=on, statement_timeout=15000 ile calisir.
Sunucunun ana surumune gore sorgu secilir; desteklenmeyen surumde sorgu calismaz (desteklenmiyor).

-- veritabani listesi (dolasilacak veritabanlari)
SELECT datname AS veritabani, has_database_privilege(datname, 'CONNECT') AS baglanir FROM pg_database WHERE datallowconn AND NOT datistemplate ORDER BY datname;

-- pg_surum (her surum)
SELECT current_setting('server_version_num')::int AS surum_num;

-- pg_sunucu_rolu (her surum)
SELECT pg_is_in_recovery() AS yedek;

-- pg_hesap (her surum)
SELECT rolsuper AS super_kullanici, pg_has_role(current_user, 'pg_read_all_stats', 'USAGE') AS pg_read_all_stats, pg_has_role(current_user, 'pg_read_all_settings', 'USAGE') AS pg_read_all_settings FROM pg_roles WHERE rolname = current_user;

-- pg_kapsam (PostgreSQL 12 ve ustu, her veritabaninda)
SELECT current_database() AS veritabani, (SELECT count(*) FROM pg_database WHERE datallowconn AND NOT datistemplate)::int AS veritabani_sayisi;

-- pg_xid_yasi (PostgreSQL 12 ve ustu)
SELECT datname, age(datfrozenxid) AS yas, mxid_age(datminmxid) AS mxid_yas, current_setting('autovacuum_freeze_max_age')::bigint AS freeze_max_age, current_setting('autovacuum_multixact_freeze_max_age')::bigint AS mxid_freeze_max_age FROM pg_database WHERE datallowconn;

-- pg_bosta_islem (PostgreSQL 12 ve ustu, yetki: pg_read_all_stats)
SELECT usename, state AS durum, extract(epoch FROM now() - xact_start)::int AS sure_sn FROM pg_stat_activity WHERE state IN ('idle in transaction','idle in transaction (aborted)') AND backend_type = 'client backend' AND pid <> pg_backend_pid() ORDER BY 3 DESC LIMIT 20;

-- pg_baglanti (PostgreSQL 12 ve ustu, yetki: pg_read_all_stats)
SELECT (SELECT count(*) FROM pg_stat_activity WHERE backend_type = 'client backend')::int AS kullanilan, current_setting('max_connections')::int AS ust;

-- pg_olu_satir (PostgreSQL 12 ve ustu, her veritabaninda)
SELECT schemaname, relname, n_live_tup AS canli, n_dead_tup AS olu, to_char(last_autovacuum AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS"Z"') AS son_autovacuum FROM pg_stat_user_tables WHERE n_dead_tup > 10000 ORDER BY n_dead_tup DESC LIMIT 20;

-- pg_gecersiz_indeks (PostgreSQL 12 ve ustu, her veritabaninda)
SELECT n.nspname AS schemaname, c.relname AS indexrelname FROM pg_index i JOIN pg_class c ON c.oid = i.indexrelid JOIN pg_namespace n ON n.oid = c.relnamespace WHERE NOT i.indisvalid LIMIT 50;

-- pg_kullanilmayan_indeks (PostgreSQL 12 ve ustu, her veritabaninda)
SELECT s.schemaname, s.indexrelname, pg_relation_size(s.indexrelid)::bigint AS bayt FROM pg_stat_user_indexes s JOIN pg_index i ON i.indexrelid = s.indexrelid WHERE s.idx_scan = 0 AND NOT i.indisunique AND NOT i.indisprimary AND pg_relation_size(s.indexrelid) > 10485760 ORDER BY 3 DESC LIMIT 20;

-- pg_sequence (PostgreSQL 12 ve ustu, her veritabaninda)
SELECT schemaname, sequencename, round((last_value::numeric / max_value::numeric), 4)::float8 AS oran FROM pg_sequences WHERE last_value IS NOT NULL AND last_value::numeric / max_value::numeric > 0.5 ORDER BY 3 DESC LIMIT 20;

-- pg_sequence_gorunurluk (PostgreSQL 12 ve ustu, her veritabaninda)
SELECT count(*)::int AS toplam, count(*) FILTER (WHERE NOT has_sequence_privilege(c.oid, 'USAGE,SELECT'))::int AS okunamayan FROM pg_class c WHERE c.relkind = 'S';

-- pg_slot (PostgreSQL 12 ve ustu)
SELECT slot_name, active, wal_status, pg_wal_lsn_diff(CASE WHEN pg_is_in_recovery() THEN pg_last_wal_receive_lsn() ELSE pg_current_wal_lsn() END, restart_lsn)::bigint AS geri_bayt FROM pg_replication_slots LIMIT 50;

-- pg_slot, PostgreSQL 12 ve oncesi
SELECT slot_name, active, NULL::text AS wal_status, pg_wal_lsn_diff(CASE WHEN pg_is_in_recovery() THEN pg_last_wal_receive_lsn() ELSE pg_current_wal_lsn() END, restart_lsn)::bigint AS geri_bayt FROM pg_replication_slots LIMIT 50;

-- pg_replikasyon (PostgreSQL 12 ve ustu, yetki: pg_read_all_stats)
SELECT application_name AS uygulama, state AS durum, sync_state AS senkron, date_part('epoch', replay_lag)::float8 AS gecikme_sn, pg_wal_lsn_diff(CASE WHEN pg_is_in_recovery() THEN pg_last_wal_receive_lsn() ELSE pg_current_wal_lsn() END, replay_lsn)::bigint AS gecikme_bayt FROM pg_stat_replication ORDER BY 5 DESC NULLS LAST LIMIT 50;

-- pg_arsiv (PostgreSQL 12 ve ustu)
SELECT current_setting('archive_mode') AS archive_mode, archived_count AS basarili, failed_count AS hatali, to_char(last_archived_time AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS"Z"') AS son_basari_utc, to_char(last_failed_time AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS"Z"') AS son_hata_utc FROM pg_stat_archiver;

-- pg_tls_parola (PostgreSQL 12 ve ustu, yetki: pg_read_all_settings)
SELECT current_setting('ssl') AS ssl, current_setting('password_encryption') AS password_encryption, current_setting('ssl_min_protocol_version') AS ssl_min_protocol_version;

-- pg_md5_roller (PostgreSQL 12 ve ustu, yetki: superuser)
SELECT count(*)::int AS sayi FROM pg_authid WHERE rolpassword LIKE 'md5%';

-- pg_super_roller (PostgreSQL 12 ve ustu)
SELECT rolname, rolcanlogin FROM pg_roles WHERE rolsuper AND rolname !~ '^pg_' LIMIT 50;

-- pg_public_create (PostgreSQL 12 ve ustu, her veritabaninda)
SELECT has_schema_privilege('public', 'public', 'CREATE') AS public_create;

-- pg_log_ayarlari (PostgreSQL 12 ve ustu)
SELECT name, setting FROM pg_settings WHERE name IN ('log_connections','log_disconnections','log_min_duration_statement');

-- pg_autovacuum (PostgreSQL 12 ve ustu)
SELECT name, setting FROM pg_settings WHERE name IN ('autovacuum', 'track_counts');

-- pg_tablo_autovacuum (PostgreSQL 12 ve ustu, her veritabaninda)
SELECT n.nspname AS schemaname, c.relname FROM pg_class c JOIN pg_namespace n ON n.oid = c.relnamespace WHERE c.relkind IN ('r', 'm') AND n.nspname NOT IN ('pg_catalog', 'information_schema') AND array_to_string(c.reloptions, ',') ~* '(^|,)autovacuum_enabled=(f|fa|fal|fals|false|n|no|of|off|0)(,|$)' ORDER BY 1, 2 LIMIT 50;

-- pg_sorgu_istatistigi (PostgreSQL 12 ve ustu, yetki: pg_read_all_settings)
SELECT current_setting('shared_preload_libraries') AS shared_preload_libraries, current_setting('pg_stat_statements.track', true) AS izleme;

-- pg_gecici_dosya (PostgreSQL 12 ve ustu)
SELECT datname, temp_files AS dosya, temp_bytes AS bayt, to_char(stats_reset AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS"Z"') AS sifirlama_utc FROM pg_stat_database WHERE datname IS NOT NULL ORDER BY temp_bytes DESC LIMIT 100;

-- pg_eklentiler (PostgreSQL 12 ve ustu, her veritabaninda)
SELECT extname, extversion FROM pg_extension;
